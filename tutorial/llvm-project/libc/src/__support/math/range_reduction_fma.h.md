// Return k and y, where
//   k = round(x * 32 / pi) and y = (x * 32 / pi) - k.
LIBC_INLINE int64_t small_range_reduction(double x, double &y) {

  y = fputil::fma<double>(x, THIRTYTWO_OVER_PI[1], y);

}
