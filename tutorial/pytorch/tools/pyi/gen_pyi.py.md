# key: (
#     slice
#     | EllipsisType
#     | None
#     | _ArrayLikeInt_co
#     | SupportsIndex
#     | tuple[slice | EllipsisType | None | _ArrayLikeInt_co | SupportsIndex, ...]

#     bool | int,

#     | _NestedSequence[_SupportsArray[_DType]]
